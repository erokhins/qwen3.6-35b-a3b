#!/usr/bin/env python3
"""tmp/w20_repro.py — scratch: does a FRESH write of the GDN dump reproduce the pinned one
bit for bit? Compares every manifest record (sha256, bytes, shape, dtype) and the byte totals.
CPU only; no clock read, no ms figure. W20 lane."""
import json
import sys
from pathlib import Path

REPO = Path("/home/erokhins/erokhins_projects/qwen3.6-35b-a3b")
PINNED = REPO / "checkpoints/dump-manifest-gdn-2026-10-26.json"
CAND = REPO / "tmp/w20_candidate_manifest.json"

a = json.loads(PINNED.read_text())
b = json.loads(CAND.read_text())


def index(m):
    out = {}
    for r in m["dumps"]:
        out[r["name"]] = r
    return out


ra, rb = index(a), index(b)
only_a = sorted(set(ra) - set(rb))
only_b = sorted(set(rb) - set(ra))
bad = []
for name in sorted(set(ra) & set(rb)):
    x, y = ra[name], rb[name]
    for key in ("sha256", "bytes", "shape", "dtype", "elements"):
        if x[key] != y[key]:
            bad.append(f"{name}.{key}: pinned={x[key]!r} fresh={y[key]!r}")

print(f"records: pinned={len(ra)} fresh={len(rb)} common={len(set(ra) & set(rb))}")
print(f"only-in-pinned={len(only_a)} only-in-fresh={len(only_b)}")
for n in only_a[:10]:
    print(f"  PINNED-ONLY {n}")
for n in only_b[:10]:
    print(f"  FRESH-ONLY {n}")
print(f"field mismatches: {len(bad)}")
for line in bad[:20]:
    print("  " + line)
print(f"bytes: pinned declared={a['bytes']['declared_records']:,} "
      f"fresh declared={b['bytes']['declared_records']:,} "
      f"equal={a['bytes']['declared_records'] == b['bytes']['declared_records']}")
print(f"counts: pinned={a['counts']} fresh={b['counts']} "
      f"equal={a['counts'] == b['counts']}")
agree_a = a["agreement_at_write_time"]
agree_b = b["agreement_at_write_time"]
print(f"reference agreement records equal={agree_a == agree_b}")
same = (not bad and not only_a and not only_b
        and a["bytes"]["declared_records"] == b["bytes"]["declared_records"]
        and a["counts"] == b["counts"])
print("REPRODUCIBLE: a fresh write reproduces the pinned dump bit for bit" if same
      else "NOT REPRODUCIBLE")
print("DONE — no timing taken")
sys.exit(0 if same else 1)
